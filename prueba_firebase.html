CONSULTAR RESULTADO DEL SERVIDOR // ======================================== function consultarResultado(accion, solicitud) { return new Promise((resolve, reject) => { const callback = "respuesta_" + crypto.randomUUID().replace(/-/g, ""); const script = document.createElement("script"); let terminado = false; function limpiar() { if (script.parentNode) { script.parentNode.removeChild(script); } delete window[callback]; } const temporizador = setTimeout(() => { if (terminado) return; terminado = true; limpiar(); reject(new Error("Tiempo de espera agotado.")); }, 10000); window[callback] = resultado => { if (terminado) return; terminado = true; clearTimeout(temporizador); limpiar(); resolve(resultado); }; const parametros = new URLSearchParams({ action: accion, solicitud: solicitud, callback: callback }); script.src = SCRIPT_URL + "?" + parametros.toString(); script.onerror = () => { if (terminado) return; terminado = true; clearTimeout(temporizador); limpiar(); reject(new Error("No se pudo consultar el servidor.")); }; document.body.appendChild(script); }); } // ======================================== // 8. ENVIAR SOLICITUD AL SERVIDOR // ========================================
